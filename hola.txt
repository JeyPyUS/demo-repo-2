def check_if_exists_s3_object(filename, bucket_name, prefix):

    # Homogeneizar el uso de las barras
    if prefix:
        prefix = prefix.strip("/ ") # Remove any spaces and slashes from the start and the beginning of the string

    s3 = boto3.client('s3')
    paginator = s3.get_paginator('list_objects_v2')
    pages = paginator.paginate(Bucket = bucket_name, Prefix = prefix)

    prefix_file = prefix + '/' + filename if prefix else filename

    for page in pages:
        for obj in page['Contents']:
            if obj['Key'] == f"{prefix_file}":
                return True

    return False
