def get_params():

    try:

        ssm=boto3.client('ssm', config=Config(proxies=proxy_definitions))

        params = {}
        params['drivername'] = 'mysql+pymysql'
        params['username'] = ssm.get_parameter(Name=aurora_username_envar)['Parameter']['Value']
        params['password'] = ssm.get_parameter(Name=aurora_password_envar, WithDecryption=True)['Parameter']['Value']
        params['host'] = ssm.get_parameter(Name=aurora_endpoint_envar)['Parameter']['Value']
        params['database'] = ssm.get_parameter(Name=aurora_database_envar)['Parameter']['Value']

        logger.debug('Successfully retrieved params from SSM')

        return params

    except Exception as e:
        logger.error('Error on get_params(): {}'.format(str(e)))

def clean_table(schema_name, table_name):

    str_truncate = "TRUNCATE TABLE " + schema_name + "." + table_name

    try:

        with myengine.connect() as conn:

            conn.execute(str_truncate)

            logger.debug('Successfully truncated table')

    except:
        logger.error("Unexpected error:", sys.exc_info()[0])
        raise

def df_to_db(schema_name, table_name, df_insert):

    try:
        logger.debug('[Inicio Insert]')

        df_insert.to_sql(name= table_name, schema= schema_name, con= myengine, if_exists='append', index= False, chunksize=1000)

        logger.debug('[Fin Insert]')

    except Exception as e:

        logger.error('Error on df_to_db(): {}'.format(str(e)))
